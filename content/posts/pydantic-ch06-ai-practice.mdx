## 01 API请求/响应模型：类型安全的通信

在AI应用中，我们天天跟各种API打交道。OpenAI、Anthropic、智谱...每个API的格式都不太一样。用Pydantic统一定义，能避免很多低级错误。

### 6.1.1 请求模型：规范输入

```python
from typing import Optional, List
from pydantic import BaseModel, Field, HttpUrl

    )

        default=1.0,
        ge=0.0,

# 使用示例
request = ChatRequest(
```

```python
import httpx
from typing import Type, TypeVar

T = TypeVar('T', bound=BaseModel)

class OpenAIClient:
    """类型安全的OpenAI客户端"""
    
    def __init__(self, api_key: str):
        self.api_key = api_key
        self.base_url = "https://api.openai.com/v1"
        self.client = httpx.Client(
            headers={'Authorization': f'Bearer {api_key}'},
            timeout=30.0
        )
    
    def create_chat_completion(
        self, 
        request: ChatRequest
    ) -> ChatResponse:
        """发送聊天请求，返回结构化响应"""
        response = self.client.post(
            f"{self.base_url}/chat/completions",
            json=request.model_dump(exclude_none=True)
        )
        response.raise_for_status()
        return ChatResponse.model_validate(response.json())
    
    def close(self):
        self.client.close()

# 使用
client = OpenAIClient(api_key="your-api-key")
try:
    chat_request = ChatRequest(
        model="gpt-3.5-turbo",
        messages=[{"role": "user", "content": "讲个笑话"}],
        temperature=0.8
    )
    result = client.create_chat_completion(chat_request)

finally:
```

## 03 提示词模板管理：类型安全的Prompt

提示词工程是AI应用的核心。用Pydantic管理提示词模板，能避免变量缺失、格式错误等问题。

### 6.3.1 提示词模板模型

```python
from typing import Dict, Any, Optional
from pydantic import BaseModel, Field, field_validator
import re

class PromptTemplate(BaseModel):
    """提示词模板"""
    name: str = Field(..., description='模板名称')
    template: str = Field(..., description='提示词模板')
    variables: Dict[str, str] = Field(
        default_factory=dict,
        description='模板变量及说明'
    )
    model: Optional[str] = Field(
        default='gpt-4',
        description='推荐使用的模型'
    )
    temperature: float = Field(
        default=0.7,
        ge=0.0,
        le=2.0,
        description='采样温度'
    )
    max_tokens: Optional[int] = Field(
        default=1000,
        gt=0,
        description='最大生成长度'
    )

    @field_validator('template')
    @classmethod
    def validate_template_vars(cls, v: str, info) -> str:
        """验证模板变量是否合法"""
        vars_in_template = re.findall(r'\{(\w+)\}', v)
        # 这里可以实现更复杂的验证逻辑
        return v

    def format(self, **kwargs) -> str:
        """格式化模板"""
        import string
        template = string.Template(self.template)
        return template.substitute(**kwargs)
    
    def format_safe(self, **kwargs) -> str:
        """安全格式化，检查变量完整性"""
        missing = set(self.variables) - set(kwargs)
        if missing:
            raise ValueError(f'缺少变量: {missing}')
        return self.format(**kwargs)

# 使用示例
summary_template = PromptTemplate(
    name='文章摘要',
    template='请将以下内容总结为${length}字以内的摘要：\n\n${content}',
    variables={
        'length': '摘要字数限制',
        'content': '需要总结的文本内容'
    },
    model='gpt-3.5-turbo',
    temperature=0.5
)

# 安全格式化
try:
    formatted = summary_template.format_safe(
        length='200',
        content='这是一篇关于Python编程的长文章...'
    )
    print(formatted)
except ValueError as e:
    print(f'错误: {e}')
```

## 04 数据解析与清洗：LLM响应的结构化

处理LLM响应时，结构化数据解析是个痛点。Pydantic能帮我们优雅解决。

### 6.4.1 从LLM响应中解析结构化数据

```python
from typing import List
from pydantic import BaseModel, Field
import json

class ExtractedEntity(BaseModel):
    """提取的实体"""
    name: str
    type: str = Field(..., description='实体类型，如：人物、地点、组织')
    confidence: float = Field(..., ge=0.0, le=1.0)

class ExtractionResult(BaseModel):
    """提取结果"""
    text: str = Field(..., description='原始文本')
    entities: List[ExtractedEntity]
    
    @classmethod
    def from_llm_json(cls, json_str: str):
        """从LLM返回的JSON字符串解析"""
        try:
            data = json.loads(json_str)
            return cls.model_validate(data)
        except json.JSONDecodeError as e:
            raise ValueError(f'JSON解析失败: {e}')
        except Exception as e:
            raise ValueError(f'数据验证失败: {e}')
    
    @classmethod
    def from_llm_text(cls, text: str):
        """从LLM返回的纯文本解析（尝试提取JSON）"""
        import re
        # 尝试找到JSON块
        json_match = re.search(r'\{.*\}', text, re.DOTALL)
        if json_match:
            return cls.from_llm_json(json_match.group())
        raise ValueError('文本中未找到有效的JSON')

# 使用示例
llm_response = '''
{
    "text": "张三在北京的清华大学工作",
    "entities": [
        {"name": "张三", "type": "人物", "confidence": 0.95},
        {"name": "北京", "type": "地点", "confidence": 0.98},
        {"name": "清华大学", "type": "组织", "confidence": 0.99}
    ]
}
'''

try:
    result = ExtractionResult.from_llm_json(llm_response)
    print(f'原始文本: {result.text}')
    for entity in result.entities:
        print(f'  - {entity.name} ({entity.type}): {entity.confidence:.2f}')
except ValueError as e:
    print(f'解析失败: {e}')
```

### 6.4.2 数据清洗

```python
from typing import List
from pydantic import BaseModel, field_validator
import re

class CleanedArticle(BaseModel):
    """清洗后的文章"""
    title: str
    content: str
    author: str = '未知'
    tags: List[str] = Field(default_factory=list)
    
    @field_validator('title', 'content', mode='after')
    @classmethod
    def strip_whitespace(cls, v: str) -> str:
        """去除首尾空白"""
        return v.strip()
    
    @field_validator('content', mode='after')
    @classmethod
    def remove_extra_newlines(cls, v: str) -> str:
        """去除多余的换行"""
        return re.sub(r'\n{3,}', '\n\n', v)
    
    @field_validator('author', mode='after')
    @classmethod
    def normalize_author(cls, v: str) -> str:
        """标准化作者名称"""
        v = v.strip()
        if not v:
            return '未知'
        # 移除常见的前缀
        v = re.sub(r'^(作者[:：]\s*)', '', v)
        return v
    
    @field_validator('tags')
    @classmethod
    def normalize_tags(cls, v: List[str]) -> List[str]:
        """标准化标签"""
        return [tag.strip().lower() for tag in v if tag.strip()]

# 使用示例
raw_article = {
    'title': '  Python 编程指南  ',
    'content': '这是一篇关于Python的文章。\n\n\n\n内容很多...\n\n\n',
    'author': '作者：张三  ',
    'tags': [' Python ', ' Programming ', '']
}

article = CleanedArticle(**raw_article)
print(f'标题: {article.title}')
print(f'作者: {article.author}')
print(f'内容预览: {article.content[:50]}...')
print(f'标签: {article.tags}')
```

## 05 API客户端封装：类型安全的HTTP客户端

封装一个类型安全的API客户端，让API调用变得简单可靠：

```python
from typing import Type, TypeVar, Generic
from pydantic import BaseModel
import httpx
from typing import Optional, Dict, Any

T = TypeVar('T', bound=BaseModel)

class APIClient(Generic[T]):
    """通用的类型安全API客户端"""
    
    def __init__(self, base_url: str, response_model: Type[T]):
        self.base_url = base_url
        self.response_model = response_model
        self.client = httpx.Client(timeout=30.0)
    
    def post(self, endpoint: str, data: BaseModel) -> T:
        """发送POST请求"""
        url = f"{self.base_url}{endpoint}"
        response = self.client.post(
            url,
            json=data.model_dump(exclude_none=True),
            headers={'Content-Type': 'application/json'}
        )
        response.raise_for_status()
        return self.response_model.model_validate(response.json())
    
    def get(self, endpoint: str, params: Optional[Dict[str, Any]] = None) -> T:
        """发送GET请求"""
        url = f"{self.base_url}{endpoint}"
        response = self.client.get(url, params=params)
        response.raise_for_status()
        return self.response_model.model_validate(response.json())
    
    def close(self):
        self.client.close()

# 使用示例
class UserRequest(BaseModel):
    name: str
    email: str

class UserResponse(BaseModel):
    id: int
    name: str
    email: str
    created_at: str

# 创建客户端
client = APIClient[UserResponse](
    base_url='https://api.example.com',
    response_model=UserResponse
)

try:
    # 创建用户
    user = client.post('/users', UserRequest(name='John', email='john@example.com'))
    print(f'创建用户: {user.name} (ID: {user.id})')
    
    # 查询用户
    user2 = client.get(f'/users/{user.id}')
    print(f'查询用户: {user2.name}')
finally:
    client.close()
```

## 06 实战：构建AI内容生成器

综合运用所有技巧，构建一个完整的AI内容生成器：

```python
from typing import List, Optional
from pydantic import BaseModel, Field, field_validator
from datetime import datetime
import json

# 1. 定义提示词模板
class ContentPrompt(BaseModel):
    topic: str
    keywords: List[str]
    tone: str = Field(default='professional', description='语气风格')
    length: int = Field(default=500, ge=100, le=2000)
    
    def to_prompt(self) -> str:
        return f"""写一篇关于'{self.topic}'的文章。
        
要求：
- 关键词: {', '.join(self.keywords)}
- 语气: {self.tone}
- 长度: 约{self.length}字

请以JSON格式返回，包含title和content字段。"""

# 2. 定义内容模型
class GeneratedContent(BaseModel):
    title: str = Field(..., min_length=5, max_length=100)
    content: str = Field(..., min_length=100)
    topic: str
    generated_at: datetime = Field(default_factory=datetime.now)
    
    @field_validator('title')
    @classmethod
    def validate_title(cls, v: str) -> str:
        if len(v.strip()) < 5:
            raise ValueError('标题太短')
        return v.strip()
    
    @field_validator('content')
    @classmethod
    def validate_content(cls, v: str) -> str:
        if len(v.strip()) < 100:
            raise ValueError('内容太短')
        return v.strip()
    
    @classmethod
    def from_llm_response(cls, topic: str, llm_response: str):
        """从LLM响应解析内容"""
        try:
            # 尝试解析JSON
            data = json.loads(llm_response)
            data['topic'] = topic
            return cls.model_validate(data)
        except json.JSONDecodeError:
            # 如果不是JSON，创建简单内容
            return cls(
                title=f'关于{topic}的内容',
                content=llm_response,
                topic=topic
            )

# 3. 定义生成器配置
class ContentGeneratorConfig(BaseModel):
    model: str = Field(default='gpt-4', description='使用的模型')
    temperature: float = Field(default=0.7, ge=0.0, le=2.0)
    max_retries: int = Field(default=3, ge=1, le=5)

# 4. 内容生成器
class ContentGenerator:
    def __init__(self, config: ContentGeneratorConfig):
        self.config = config
    
    def generate(self, prompt: ContentPrompt) -> GeneratedContent:
        """生成内容（伪代码，实际需要调用LLM API）"""
        print(f"正在生成关于'{prompt.topic}'的内容...")
        print(f"提示词: {prompt.to_prompt()}")
        
        # 模拟LLM响应
        mock_response = json.dumps({
            "title": f"深入探讨{prompt.topic}",
            "content": f"这是一篇关于{prompt.topic}的详细文章。"
                       f"关键词包括：{', '.join(prompt.keywords)}。"
                       f"本文将全面分析{prompt.topic}的各个方面..."
        })
        
        # 解析响应
        content = GeneratedContent.from_llm_response(
            topic=prompt.topic,
            llm_response=mock_response
        )
        
        return content

# 使用示例
config = ContentGeneratorConfig(
    model='gpt-4',
    temperature=0.8
)

generator = ContentGenerator(config)

prompt = ContentPrompt(
    topic='Pydantic在AI开发中的应用',
    keywords=['Pydantic', 'AI', '数据验证', '类型安全'],
    tone='technical',
    length=800
)

content = generator.generate(prompt)
print(f"\n生成成功!")
print(f"标题: {content.title}")
print(f"发布时间: {content.generated_at}")
print(f"内容长度: {len(content.content)}字")
```
